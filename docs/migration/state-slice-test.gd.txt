## A Host, a Guest and a late Guest in one process, each machine its own subtree with its own
## SceneMultiplayer, talking real ENet on localhost. Prints PASS or FAIL with every check.
extends SceneTree

const PORT := 24611
var _fails := 0
var _checks := 0


func _initialize() -> void:
	_run.call_deferred()


func _run() -> void:
	var host := _machine("Host")
	_check(host["session"].host(PORT) == OK, "the Host listens on %d" % PORT)
	var guest := _machine("Guest")
	_check(guest["session"].join("127.0.0.1", PORT) == OK, "the Guest starts joining")
	await _until(func() -> bool: return guest["welcomed"][0], "the Guest is welcomed")
	var g_id: int = guest["session"].me()
	var door := &"Hall/Door/Openable"
	var radio := &"Hall/Radio/Carriable"

	# The scene's own state shows in the snapshot, and the locked door is locked everywhere.
	_check(host["parts"].of(&"Hall/Cellar/Openable").locked, "the Cellar Door is locked (set in level.tscn)")

	# 1. The Guest clicks the Door: before the answer nothing changed on the Guest.
	(guest["parts"].of(door) as Openable).use(g_id)
	_check(not guest["parts"].of(door).open, "the Guest's Door is still shut before the Host answers")
	await _until(func() -> bool: return guest["parts"].of(door).open, "the Guest's Door opens")
	_check(host["parts"].of(door).open, "the Host's Door is open")

	# 2. The Guest picks up the Radio; the Guest asking for a second thing is refused.
	(guest["parts"].of(radio) as Carriable).pick(g_id)
	await _until(func() -> bool: return guest["parts"].of(radio).holder == g_id, "the Guest holds the Radio")
	_check(host["parts"].of(radio).holder == g_id, "the Host sees the Guest holding the Radio")

	# 3. A locked Door refuses, on the Host's word.
	var refusals: Array[String] = []
	guest["parts"].applied.connect(func(ev: Dictionary) -> void:
		if not ev["ok"]:
			refusals.append(ev["reason"]))
	(guest["parts"].of(&"Hall/Cellar/Openable") as Openable).use(g_id)
	await _until(func() -> bool: return refusals.has("locked"), "the Guest hears 'locked'")

	# 4. A second Guest joins late: it gets the open Door and the held Radio in its snapshot.
	var late := _machine("Late")
	_check(late["session"].join("127.0.0.1", PORT) == OK, "the late Guest starts joining")
	await _until(func() -> bool: return late["welcomed"][0], "the late Guest is welcomed")
	_check(late["parts"].of(door).open, "the late Guest sees the Door open")
	_check(late["parts"].of(radio).holder == g_id, "the late Guest sees who holds the Radio")
	_check(is_equal_approx((late["level"].get_node("Hall/Door/Hinge") as Node3D).rotation.y, PI / 2),
		"the late Guest's Door stands open at once, without a swing")

	# 5. The late Guest tries to take the Radio: first come wins, refused "gone".
	var l_id: int = late["session"].me()
	(late["parts"].of(radio) as Carriable).pick(l_id)
	await _frames(20)
	_check(host["parts"].of(radio).holder == g_id, "the Radio stays with the first Guest")

	# 6. Every machine has the same state.
	await _frames(10)
	var snap: Dictionary = host["parts"].snapshot()
	_check(snap == guest["parts"].snapshot() and snap == late["parts"].snapshot(), "all three machines agree")
	print("SNAPSHOT ", var_to_str(snap).replace("\n", " "))

	# 7. The first Guest leaves holding the Radio: the Host lets it go, everyone sees it lie.
	guest["session"].multiplayer.multiplayer_peer.close()
	await _until(func() -> bool: return late["parts"].of(radio).holder == 0, "the late Guest sees the Radio dropped")
	_check(host["parts"].of(radio).holder == 0, "the Host dropped the Radio of the Guest who left")

	print("%s %d checks, %d failed" % ["PASS" if _fails == 0 else "FAIL", _checks, _fails])
	quit(1 if _fails else 0)


func _machine(machine_name: String) -> Dictionary:
	var m := Node.new()
	m.name = machine_name
	root.add_child(m)
	set_multiplayer(SceneMultiplayer.new(), m.get_path())
	var level: Node3D = load("res://game/house/level.tscn").instantiate()
	m.add_child(level)
	var parts := Parts.new()
	parts.name = "Parts"
	parts.level = level
	m.add_child(parts)
	var session := Session.new()
	session.name = "Session"
	session.parts = parts
	m.add_child(session)
	var welcomed := [false]
	session.welcomed.connect(func() -> void: welcomed[0] = true)
	return {"node": m, "level": level, "parts": parts, "session": session, "welcomed": welcomed}


func _until(cond: Callable, what: String, max_frames := 600) -> void:
	for i in max_frames:
		if cond.call():
			_check(true, what)
			return
		await process_frame
	_check(false, what + " (timed out)")


func _frames(n: int) -> void:
	for i in n:
		await process_frame


func _check(ok: bool, what: String) -> void:
	_checks += 1
	if not ok:
		_fails += 1
	print("  %s %s" % ["ok  " if ok else "FAIL", what])
